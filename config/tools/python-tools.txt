# igraph — the `communities` extra of code-review-graph. Without it, community
# detection falls back to grouping by directory, which only restates the file
# tree; with it, Leiden runs and the groupings are semantic. Measured on
# mtg-graph: 7 directory groups (cohesion 0.115-0.295) became 16 communities
# (0.101-0.545), cutting across directories.
# Listed as its own package rather than as `code-review-graph[communities]` so
# that --check verifies it directly. bare_name() strips the brackets, so an
# extra is only ever checked through its parent distribution's version: igraph
# could vanish from the environment, or drift off the lock, and
# `code-review-graph[communities]` would still report green. Named on its own
# line it is pinned and checked in its own right.
igraph

# zotero-mcp-server — 54yyyu/zotero-mcp. NOT the PyPI package named
# "zotero-mcp", which is the unrelated kujenga/zotero-mcp project.
# Provides the `zotero-cli` binary and `zotero-mcp install-skill`.
# The MCP server itself is deliberately NOT enabled. Its default profile's schemas
# cost ~13.4k tokens per request whenever tool search is off (with tool search on,
# only tool names are sent), versus ~98 for the CLI skill in any configuration.
# See "Why capabilities arrive as CLI skills" in config/README.md.
# No [semantic] extra — it pulls torch via sentence-transformers.
zotero-mcp-server

# opencite — academic literature search / citation / PDF retrieval CLI.
# Used by the `opencite` skill (opencite@research-skills marketplace plugin).
# The [pdf] extra is required for `opencite pdf`, `convert`,
# `batch-fetch --convert`, and preprint HTML full-text; without it those
# subcommands fail outright. It costs markitdown → magika → onnxruntime + numpy
# (~250MB). Taken anyway, unlike zotero-mcp-server's [semantic] extra above,
# because it gates core function rather than an optional search mode.
opencite[pdf]

# mistralai<2 — a constraint, not a tool. markit-mistral 0.2.3 asks for
# `mistralai>=1.0.0` with no upper bound, but imports `DocumentURLChunk` from it,
# which mistralai 2.x removed. Unconstrained, uv resolves 2.x and
# `opencite convert --converter mistral` dies on ImportError. opencite imports
# markit-mistral lazily and only when MISTRAL_API_KEY is set, so the failure is
# invisible until the day a key is added — hence the pin rather than a note.
mistralai<2
